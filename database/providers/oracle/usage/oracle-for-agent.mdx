Use `OracleDb` to persist an Agent's conversation history.

```python oracle_for_agent.py
import os

from agno.agent import Agent
from agno.db.oracle import OracleDb
from agno.models.openai import OpenAIResponses


def main():
    db = OracleDb(db_url=os.environ["ORACLE_DB_URL"], create_schema=False)
    try:
        agent = Agent(
            model=OpenAIResponses(id=os.environ["OPENAI_MODEL"]),
            db=db,
            user_id="alice",
            session_id="oracle-history-example",
            add_history_to_context=True,
            num_history_runs=3,
        )
        agent.print_response("My project is called Cedar.")
        agent.print_response("What is my project called?")
    finally:
        db.close()


if __name__ == "__main__":
    main()
```

Both calls share a user and session. `add_history_to_context=True` includes prior runs in later requests. Reuse the database configuration and IDs to continue after a restart.
