Persist an Agent's conversation history with `AsyncOracleDb` and awaited calls.

```python async_oracle_for_agent.py
import asyncio
import os

from agno.agent import Agent
from agno.db.oracle import AsyncOracleDb
from agno.models.openai import OpenAIResponses


async def main():
    db = AsyncOracleDb(db_url=os.environ["ORACLE_DB_URL"], create_schema=False)
    try:
        agent = Agent(
            model=OpenAIResponses(id=os.environ["OPENAI_MODEL"]),
            db=db,
            user_id="alice",
            session_id="async-oracle-history-example",
            add_history_to_context=True,
            num_history_runs=3,
        )
        await agent.aprint_response("My project is called Cedar.")
        await agent.aprint_response("What is my project called?")
    finally:
        await db.close()


if __name__ == "__main__":
    asyncio.run(main())
```

Reuse the user and session IDs to reload conversation history. Keep construction, requests, and cleanup in one event loop.
